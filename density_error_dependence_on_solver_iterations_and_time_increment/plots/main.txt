fn main() -> Result<(), Box<dyn std::error::Error>> {
    // parse args
    let args = Args::parse();

    init_logging(&args);

    let resolution = (1600, 1200);
    let filled = false;
    let line_thickness = 2;

    let out_path = std::path::Path::new(&args.out_folder);
    folder_exists_or_create(out_path).unwrap();

    // Load measurements into DataFrame
    let df_plotter = DFPlotter::from_measurement_folder(args.in_folder)?;

    println!("Loaded DataFrame:\n{}", df_plotter.head_of_df(Some(5)));

    let mut default_parameters: HashMap<String, MeasurementType> = HashMap::new();
    // map.insert(PlotParams::Time.to_key(), MeasurementType::Float(10));
    // map.insert(PlotParams::Density.to_key(), MeasurementType::Float(10));
    // map.insert(PlotParams::DensityError.to_key(), MeasurementType::Float(10));
    // map.insert(PlotParams::KineticEnergy.to_key(), MeasurementType::Float(10));
    // map.insert(PlotParams::Stiffness.to_key(), MeasurementType::Float(15000000.));
    default_parameters.insert(PlotParams::FluidViscosity.to_key(), MeasurementType::Float(0.2));
    default_parameters.insert(PlotParams::BoundaryViscosity.to_key(), MeasurementType::Float(0.2));
    default_parameters.insert(PlotParams::FluidDepth.to_key(), MeasurementType::Float(20.249999999999996));
    default_parameters.insert(PlotParams::RestDensityGridSpacing.to_key(), MeasurementType::Float(0.8));
    default_parameters.insert(PlotParams::SmoothingLength.to_key(), MeasurementType::Float(0.8));
    default_parameters.insert(PlotParams::RestDensity.to_key(), MeasurementType::Float(900.));
    default_parameters.insert(PlotParams::TimeStepSize.to_key(), MeasurementType::Float(0.01));
    default_parameters.insert(PlotParams::SolverIterations.to_key(), MeasurementType::UInt(25));

    let fixed_parameters = vec![
        // PlotParams::Time,
        // PlotParams::Density,
        // PlotParams::DensityError,
        // PlotParams::KineticEnergy,
        // PlotParams::Stiffness,
        PlotParams::FluidViscosity,
        PlotParams::BoundaryViscosity,
        PlotParams::FluidDepth,
        PlotParams::RestDensityGridSpacing,
        PlotParams::SmoothingLength,
        PlotParams::RestDensity,
        PlotParams::TimeStepSize,
        PlotParams::SolverIterations,
    ];

    let dffilter = DFFilter::new_with_fixed_parameters(default_parameters, fixed_parameters);
    
    if let Err(e) = df_plotter.plot(
        &PlotParams::Time,
        &PlotParams::DensityError,
        &PlotParams::TimeStepSize,
        dffilter.clone(),
        resolution,
        5.0..15.0,
        -0.01..0.335,
        filled,
        line_thickness,
        out_path,
        "_solver_iterations_25"
    ) {
        error!("Failed to create plot: {e}");
    }

    if let Err(e) = df_plotter.plot(
        &PlotParams::Time,
        &PlotParams::KineticEnergy,
        &PlotParams::TimeStepSize,
        dffilter,
        resolution,
        5.0..15.0,
        180.0..250.0,
        filled,
        line_thickness,
        out_path,
        "_solver_iterations_25"
    ) {
        error!("Failed to create plot!");
    }
    
    if let Err(e) = df_plotter.plot(
        &PlotParams::SolverIterations,
        &PlotParams::MeanDensityError,
        &PlotParams::TimeStepSize,
        dffilter.clone(),
        resolution,
        5.0..25.0,
        -0.01..0.335,
        filled,
        line_thickness,
        out_path,
        ""
    ) {
        error!("Failed to create plot: {e}");
    }
    
    if let Err(e) = df_plotter.plot3d(
        &PlotParams::TimeStepSize,
        &PlotParams::SolverIterations,
        &PlotParams::MeanDensityError,
        dffilter.clone(),
        resolution,
        0.001..0.01,
        5.0..25.0,
        -0.1..0.4,
        filled,
        line_thickness,
        out_path,
        ""
    ) {
        error!("Failed to create plot: {e}");
    }

    Ok(())
}
